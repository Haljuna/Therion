# Orbe HUD Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Trocar o visual da orbe holográfica do Therion pela Orbe HUD (esfera de 3200 pontos roxo e ciano, núcleo, órbitas, eixo e moldura com nome do estado), com aparência própria para cada um dos 13 estados.

**Architecture:** O motor atual (`OrbGL`: cena em FBO com MSAA, rastro, bloom em escalas, `QualityController`, cadência por estado, modo reduzido) fica. Trocam a geometria (`orb_geometry.py`), os shaders da cena (`orb_shaders.py`) e os canais de movimento (`orb_motion.py`). Dados puros (cores, nomes, aparência por estado, proporções) ficam num módulo novo `hud_style.py`; a moldura é desenhada com `QPainter` num módulo novo `hud_frame.py`, usado tanto pelo `OrbGL` (depois do OpenGL) quanto pela reserva 2D.

**Tech Stack:** Python 3.14, PySide6 (`QOpenGLWidget`, OpenGL 2.1 compatível, GLSL estilo `attribute`/`varying`), numpy, pytest.

**Spec:** `docs/superpowers/specs/2026-11-13-orbe-hud-design.md`. Protótipo de referência (fórmulas de movimento e desenho da moldura): `docs/superpowers/specs/2026-11-13-orbe-hud-prototipo.html`, funções `draw()` e `drawFrame()`, paleta `mix`.

from PySide6.QtGui import QImage, QPainter, QColor
from therion.interface.hud_frame import FrameState, paint_hud_frame
from therion.interface.hud_style import hud_layout

def frame(**kw):
    base = dict(label='OUVINDO', level=.5, arc_deg=68, arc_scale=1, arc_rotation=0, tick_rotation=0,
                dash_rotation=0, blink=1, amber=0, voice_ring=0, voice=0)
    base.update(kw); return FrameState(**base)

def render(w, h, f, dpr=1.):
    img = QImage(int(w*dpr), int(h*dpr), QImage.Format_ARGB32_Premultiplied); img.setDevicePixelRatio(dpr)
    img.fill(QColor(0, 0, 0, 0)); p = QPainter(img)
    paint_hud_frame(p, hud_layout(0, 0, w, h), f); p.end(); return img

def lit_box(img):
    xs, ys = [], []
    for y in range(img.height()):
        for x in range(img.width()):
            if img.pixelColor(x, y).alpha() > 20: xs.append(x); ys.append(y)
    return min(xs), min(ys), max(xs), max(ys)

def test_frame_draws_something(app):
    assert lit_box(render(400, 400, frame()))

def test_frame_fits_inside_small_area(app):
    for dpr in (1., 1.5):
        img = render(160, 160, frame(label='AGUARDANDO RESPOSTA'), dpr)
        x0, y0, x1, y1 = lit_box(img)
        assert x0 >= 0 and y0 >= 0 and x1 < img.width() and y1 < img.height() - 1

def test_amber_frame_is_warmer(app):
    def mean_red_minus_blue(img):
        s = n = 0
        for y in range(0, img.height(), 2):
            for x in range(0, img.width(), 2):
                c = img.pixelColor(x, y)
                if c.alpha() > 40: s += c.red() - c.blue(); n += 1
        return s / max(1, n)
    assert mean_red_minus_blue(render(400, 400, frame(amber=1))) > mean_red_minus_blue(render(400, 400, frame())) + 60

def test_painter_state_is_restored(app):

    p.setOpacity(.3); paint_hud_frame(p, hud_layout(0, 0, 200, 200), frame())
    assert abs(p.opacity() - .3) < 1e-9; p.end()
